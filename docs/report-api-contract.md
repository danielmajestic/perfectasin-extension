# Report Generator API Contract
**Version:** 1.0 | **Date:** 2026-03-22
**Status:** APPROVED — Sam can build against this contract with mock data

---

## Endpoint

```
POST /api/v1/report/generate
```

**Auth:** Firebase ID token in `Authorization: Bearer <token>` header (same as all existing endpoints)
**Timeout:** 60 seconds max

---

## Request Schema

```typescript
interface GenerateReportRequest {
  asin: string;                    // e.g. "B0B44TJQZX"
  marketplace: string;             // "US" | "UK" | "DE" | "FR" | "IT" | "ES" | "CA" | "JP"
  format: "pdf" | "html" | "both"; // What to generate

  // Product data — Sam already has all of this from the existing tab scrapers
  scrapedData: {
    // Title tab data
    title: string;
    category?: string | null;
    brand?: string | null;
    competitors?: CompetitorTitle[];    // reuse existing CompetitorTitle type

    // Bullets tab data
    bulletPoints: string[];             // array of bullet point strings

    // Description tab data
    description: string;               // raw description text (plain or HTML)
    hasAplus?: boolean;                 // A+ content detected

    // Hero Image tab data
    heroImageUrl?: string | null;       // main product image URL
    heroHiresUrl?: string | null;       // high-res zoom URL
    zoomEligible?: boolean;
    heroAlt?: string | null;
    imageCount?: number;                // total static images in gallery
    videoCount?: number;
    hasVideo?: boolean;
    has360?: boolean;
    galleryAltTexts?: string[];

    // Price tab data
    currentPrice?: string | null;
    currentPriceNumeric?: number | null;
    listPrice?: string | null;
    listPriceNumeric?: number | null;
    dealBadgeText?: string | null;
    couponText?: string | null;
    subscribeSavePrice?: string | null;
    buyBoxStatus?: string;              // "winning" | "competing" | "suppressed" | "unknown"
    rating?: number | null;
    reviewCount?: string | null;
    priceCompetitors?: AnalyzePriceCompetitor[]; // reuse existing type
    keywordUsed?: string | null;
  };
}
```

**Note on field naming:** Request uses camelCase (JS convention) since it comes from the extension. Backend converts to snake_case internally. This matches how existing endpoints work.

## Response Schema

```typescript
interface GenerateReportResponse {
  reportId: string;                // unique ID, format: "rpt_{uuid}"
  overallGrade: string;            // "A" | "B" | "C" | "D" | "F"
  overallScore: number;            // 0-100 numeric score

  // Executive summary (generated by synthesis prompt)
  executiveSummary: {
    biggestStrength: string;       // 1-2 sentences
    biggestWeakness: string;       // 1-2 sentences
    estimatedRevenueImpact: string; // e.g. "Optimizing title and bullets could increase conversion 15-25%"
  };

  // Per-section results — each mirrors the existing analysis response
  // but with an added "status" field for partial failure handling
  sections: {
    title: {
      status: "success" | "error";
      error?: string;              // only if status === "error"
      score: number;               // overall_score from AnalysisResponse
      grade: string;               // A-F letter grade
      currentTitle: string;
      optimizedTitle?: string;     // best variation title
      issues: string[];            // compliance issue messages
      strengths: string[];
      weaknesses: string[];
      recommendations: string[];
      characterCount: number;
      mobileTruncated: boolean;
    };

    bullets: {
      status: "success" | "error";
      error?: string;
      score: number;
      grade: string;
      bulletScores: Array<{
        index: number;
        text: string;
        overall: number;
        feedback: string;
        isLegal?: boolean;
      }>;
      optimizedBullets?: string[];  // best variation bullet set
      strengths: string[];
      weaknesses: string[];
      recommendations: string[];
    };

    description: {
      status: "success" | "error";
      error?: string;
      score: number;
      grade: string;
      optimizedDescription?: string;  // best variation (plain text)
      charCount: number;
      complianceFlag?: string | null;
      aplusDetected: boolean;
      strengths: string[];
      weaknesses: string[];
      recommendations: string[];
    };

    heroImage: {
      status: "success" | "error";
      error?: string;
      score: number;
      grade: string;
      dimensions: Array<{
        label: string;
        score: number;
        finding: string;
        recommendation: string;
      }>;
      criticalIssues: string[];
      quickWins: string[];
      recommendations: string[];
      zoomEligible: boolean;
      imageCount: number;
      hasVideo: boolean;
    };

    price: {
      status: "success" | "error";
      error?: string;
      score: number;
      grade: string;
      pricePercentile?: number | null;
      marketMedian?: number | null;
      marketMin?: number | null;
      marketMax?: number | null;
      competitorCount: number;
      buyBoxStatus?: string;
      quickWins: string[];
      recommendations: string[];
      priceRecommendation?: {
        suggestedPrice: string;
        confidence: "high" | "medium" | "low";
        rationale: string;
        expectedImpact: string;
      } | null;
    };
  };

  // Action plan (generated by synthesis prompt)
  actionPlan: Array<{
    priority: number;              // 1 = highest priority
    action: string;                // what to do
    section: string;               // "title" | "bullets" | "description" | "heroImage" | "price"
    impact: "high" | "medium" | "low";
    difficulty: "easy" | "medium" | "hard";
    estimatedLift?: string;        // e.g. "+5-10% CTR"
  }>;

  // Copy blocks — all optimized text in one place for easy copy-paste
  copyBlocks: Array<{
    fieldLabel: string;            // Seller Central field name, e.g. "Product Title", "Bullet Point 1"
    content: string;               // the optimized text
    section: string;               // which report section this belongs to
  }>;

  // Generated report files
  htmlContent: string;             // full self-contained HTML string
  pdfBase64: string | null;        // base64-encoded PDF, or null if PDF generation failed

  // Metadata
  generatedAt: string;             // ISO 8601 timestamp
  processingTimeMs: number;        // total time for all analyses + synthesis
  modulesCompleted: number;        // how many of 5 analyses succeeded (0-5)
  modulesTotal: 5;
}
```

## Error Responses

### 401 Unauthorized
```json
{ "error": "Missing or invalid Firebase ID token" }
```

### 422 Validation Error
```json
{ "error": "Validation error", "detail": "ASIN must be 10 alphanumeric characters" }
```

### 429 Rate Limited (Phase 2 — not enforced in Phase 1)
```json
{
  "error": "Free tier limited to 1 report per month. Upgrade to Pro for unlimited reports.",
  "upgradeUrl": "https://perfectasin.com/pricing"
}
```

### 500 Server Error
```json
{ "error": "Report generation failed", "detail": "..." }
```

### 504 Timeout
```json
{ "error": "Report generation timed out. Please try again." }
```

## Extension Integration Notes for Sam

1. **New API method to add to `utils/api.ts`** (when wiring to real endpoint):
```typescript
async generateReport(data: GenerateReportRequest): Promise<GenerateReportResponse> {
  return this.post<GenerateReportResponse>('/api/v1/report/generate', data);
}
```

2. **Data Sam already has available** from existing tab scrapers — no new scraping needed:
   - Title, brand, category, bullets → from content script (used by Title + Bullets tabs)
   - Description, hasAplus → from content script (used by Description tab)
   - Hero image data → from content script (used by Hero Image tab)
   - Price + competitors → from content script + SERP scraper (used by Price tab)

3. **The "Generate Full Report" button** collects all scraped data into one `GenerateReportRequest` and fires a single POST. The backend runs all 5 analyses — Sam does NOT need to call 5 separate endpoints.

4. **PDF download:** Decode `pdfBase64` and trigger browser download:
```typescript
const blob = new Blob([Uint8Array.from(atob(response.pdfBase64), c => c.charCodeAt(0))], { type: 'application/pdf' });
const url = URL.createObjectURL(blob);
chrome.downloads.download({ url, filename: `PerfectASIN-Report-${response.reportId}.pdf` });
```

5. **HTML view:** Open in new tab:
```typescript
const blob = new Blob([response.htmlContent], { type: 'text/html' });
const url = URL.createObjectURL(blob);
chrome.tabs.create({ url });
```

## v3.7.5 Addendum — "July 27 Readiness" (2026-07-11, corrected 2026-07-11 post-QA)

**Status: RECONCILED AGAINST REAL BACKEND SOURCE** (`app/models/report_models.py`), not just the original design intent below. Two corrections from a QA bug-fix pass — see `~/agents/sam/HANDOFF.md` "v3.7.5 QA bug-fix pass" for the full story:

1. **`sections.title` does NOT carry `currentTitle` or `characterCount`** (or `optimizedTitle`/`mobileTruncated`/`strengths`/`weaknesses`/`recommendations`, contra this doc's original pre-v3.7.5 text above). The real `TitleSectionResult` only has `score`, `grade`, `issues: string[]`, `optimizedContent: string | string[] | null`, `sectionHeading`, plus the 3 Item Highlight fields below. **The extension gets title text/length from its own scrape, not from the report response** — the backend never needs to echo it back.
2. **`backendFields` is nested under `sections.backendFields`**, not top-level on `ReportJson`. It's a sibling field on the `ReportSections` model, same as `title`/`bullets`/etc.

### `sections.title` — real shape
```typescript
{
  score: number;
  grade: string;
  issues: string[];
  optimizedContent?: string | string[] | null;  // string for title, string[] for bullets/description
  sectionHeading?: string;
  // v3.7.5 Item Highlight fields — confirmed correct, no changes needed
  currentItemHighlight?: string | null;   // scraped/existing Item Highlight, if any
  optimizedItemHighlight?: string;        // AI-generated, <=125 chars, plain text, must NOT duplicate title wording
  itemHighlightCharCount?: number;
}
```

### `sections.backendFields` object (nested, not top-level)
```typescript
// path: reportJson.sections.backendFields, NOT reportJson.backendFields
interface BackendFieldsSection {
  status: 'success' | 'error';
  error?: string;
  genericKeywords: string;             // optimized string, hard-capped 249 BYTES (UTF-8), not chars
  genericKeywordsByteCount: number;
  seasonalSwapString: string;          // secondary alternate string, also <=249 bytes
  seasonalSwapByteCount: number;
  bullets: string[];                   // MIN 5 per ASIN — extension renders whatever length is sent, no client-side padding
  productDescription: string;
}
```
Confirmed (Dan, v3.7.5 QA): on generation failure the whole `sections.backendFields` key is **absent** (not present with `status: 'error'`) — matches every other section's existing failure pattern. Extension renders nothing when absent; does not branch on `status === 'error'` as a display state.

### New top-level `compliance` object
```typescript
type ComplianceRiskTier = 'banned' | 'high_risk' | 'caution';

interface ComplianceViolation {
  term: string;
  field: string;   // 'Item Name (Title)' | 'Item Highlight' | 'Bullet Point' | 'Product Description' | 'Generic Keywords (Search Terms)'
  riskTier: ComplianceRiskTier;
  agency: string;  // e.g. "FDA (cosmetic vs drug line)"
  context: string; // explanation from amazon-compliance-keywords-db.json
}

interface ComplianceResult {
  status: 'compliant' | 'violations_found';
  violations: ComplianceViolation[];
  disclaimer: string;  // use amazon-compliance-keywords-db.json -> metadata.disclaimer_copy verbatim
}
```
Pass/fail is independent of `overallGrade` — a report can be grade A and `compliance.status === 'violations_found'`.
